# SECTION 0: ONE-TIME CLI SETUP COMMANDS
# ==============================================================================
# Run these ONCE on a fresh machine before anything else.
# Copy-paste each block into your terminal in order.
# ------------------------------------------------------------------------------

# ── Step 0.1  Install Node.js (via nvm — recommended) ────────────────────────
#   curl -o- https://raw.githubusercontent.com/nvm-sh/nvm/v0.39.7/install.sh | bash
#   source ~/.zshrc          # or ~/.bashrc on Linux
#   nvm install 18
#   nvm use 18
#   node -v                  # should print v18.x.x
#   npm -v                   # should print 9.x.x or higher

# ── Step 0.2  Install Flutter SDK ─────────────────────────────────────────────
#   # macOS (via Homebrew)
#   brew install --cask flutter
#
#   # OR download directly:
#   #   https://flutter.dev/docs/get-started/install
#
#   flutter --version        # should print Flutter 3.x.x
#   flutter doctor           # fix any missing dependencies shown here

# ── Step 0.3  Install Firebase CLI ────────────────────────────────────────────
#   npm install -g firebase-tools
#   firebase --version       # verify install
#   firebase login           # opens browser for Google auth

# ── Step 0.4  Install FlutterFire CLI (generates firebase_options.dart) ───────
#   dart pub global activate flutterfire_cli
#   # Add Dart pub cache to PATH if not already:
#   export PATH="$PATH":"$HOME/.pub-cache/bin"

# ── Step 0.5  Install Google Cloud CLI (for Pub/Sub topic creation) ───────────
#   # macOS
#   brew install --cask google-cloud-sdk
#   gcloud init              # follow prompts to authenticate + select project
#   gcloud pubsub topics create emergency-packets

# ── Step 0.6  Link Firebase project ───────────────────────────────────────────
#   firebase use sync-bridge-f88c2
#   # OR if using your own project:
#   firebase use YOUR_PROJECT_ID

# ── Step 0.7  Configure Flutter + Firebase (regenerates firebase_options.dart) ─
#   cd mobile-app
#   flutterfire configure --project=sync-bridge-f88c2
#   cd ..

# ── Step 0.8  Verify full Flutter environment ──────────────────────────────────
#   flutter doctor -v
